import getpass, json, re, subprocess, sys, time, urllib.parse, urllib.request
from pathlib import Path

API_KEY = "AIzaSyDFdPaMtvNcyP_ZFecXD1tKgP-XFhm6uHw"
DB_URL = "https://bothost-5a1e7-default-rtdb.firebaseio.com"

process = None

def api(url, method="GET", payload=None):
    data = None if payload is None else json.dumps(payload).encode()
    req = urllib.request.Request(url, data=data, method=method, headers={"Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=25) as r:
        raw = r.read().decode()
        return json.loads(raw) if raw else None

def db_url(path, token):
    encoded = "/".join(urllib.parse.quote(p, safe="") for p in path.split("/"))
    return f"{DB_URL}/{encoded}.json?auth={urllib.parse.quote(token)}"

def db_get(path, token):
    return api(db_url(path, token))

def db_patch(path, token, payload):
    return api(db_url(path, token), "PATCH", payload)

def db_delete(path, token):
    return api(db_url(path, token), "DELETE")

def log(uid, token, text):
    try:
        api(db_url(f"logs/{uid}", token), "POST", {"message": text, "createdAt": {".sv": "timestamp"}})
    except Exception:
        pass

def set_project_runtime(uid, token, project_id, status, live_url=""):
    if not uid or not token or not project_id:
        return
    try:
        db_patch(f"projects/{uid}/{project_id}", token, {

            "liveUrl": live_url,
            "updatedAt": {".sv": "timestamp"},

def sign_in():
    email = input("Firebase email: ").strip()

        "https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key=" + API_KEY,

    )

def refresh_token(refresh):
    r = api(
        "https://securetoken.googleapis.com/v1/token?key=" + API_KEY,

        {"grant_type": "refresh_token", "refresh_token": refresh},

def safe_project(project_id, project):

        size = len(content.encode("utf-8"))
        if size > 200 * 1024:

def start_tunnel(port=5000):
    global tunnel_process
    stop_tunnel()
    try:
        tunnel_process = subprocess.Popen(
            ["cloudflared", "tunnel", "--url", f"http://127.0.0.1:{port}"],
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
            text=True,
        )
        deadline = time.time() + 25
        url_pattern = re.compile(r"https://[a-zA-Z0-9.-]+\\.trycloudflare\\.com")
        while time.time() < deadline:
            if tunnel_process.poll() is not None:
                break
            line = tunnel_process.stdout.readline() if tunnel_process.stdout else ""
            if line:
                match = url_pattern.search(line)
                if match:
                    return match.group(0)
            time.sleep(0.1)
    except FileNotFoundError:
        print("cloudflared পাওয়া যায়নি।")
    except Exception as exc:
        print("Tunnel error:", exc)
    return ""

def stop_tunnel():
    global tunnel_process
    if tunnel_process and tunnel_process.poll() is None:
        tunnel_process.terminate()
        try:
            tunnel_process.wait(timeout=5)
        except subprocess.TimeoutExpired:
            tunnel_process.kill()
            tunnel_process.wait()
    tunnel_process = None

def process_commands(uid, token):
    global process, active_project, active_token
    active_token = token
